#!/usr/bin/env node

/**
 * Coverage analysis for changed files — MetaMask Extension
 * Injected by farmslot at prepare time. Self-contained, no external deps.
 *
 * Usage:
 *   node scripts/coverage-analyze.js                    # Analyze all PR changes vs main
 *   node scripts/coverage-analyze.js --files src/a.ts   # Analyze specific files
 */

const { execSync } = require('child_process');
const fs = require('fs');
const path = require('path');

const COVERAGE_THRESHOLD = 80;
const COVERAGE_DIR = path.join(process.cwd(), 'coverage', 'unit');

// ---------------------------------------------------------------------------
// Git helpers (inlined — no git-utils dependency)
// ---------------------------------------------------------------------------

function git(args) {
  return execSync(`git ${args}`, { encoding: 'utf8', cwd: process.cwd() }).trim();
}

function getChangedFilesVsMain() {
  const raw = git('diff main --name-only --diff-filter=d');
  return raw
    .split('\n')
    .filter(Boolean)
    .filter(f => /\.(ts|tsx|js|jsx)$/.test(f))
    .filter(f => !f.includes('.test.') && !f.includes('.spec.') && !f.includes('.stories.'));
}

function getChangedLines(file) {
  try {
    const diff = git(`diff --unified=0 main...HEAD -- "${file}"`);
    const lines = [];
    for (const line of diff.split('\n')) {
      const m = line.match(/^@@ -\d+(?:,\d+)? \+(\d+)(?:,(\d+))? @@/);
      if (m) {
        const start = parseInt(m[1]);
        const count = m[2] ? parseInt(m[2]) : 1;
        for (let i = start; i < start + count; i++) lines.push(i);
      }
    }
    return lines.sort((a, b) => a - b);
  } catch {
    return [];
  }
}

function hasTestFile(file) {
  const testFile = file.replace(/\.(ts|tsx|js|jsx)$/, '.test.$1');
  return fs.existsSync(testFile);
}

function getBranchName() {
  return git('branch --show-current').replace(/[/\\:*?"<>|]/g, '-');
}

// ---------------------------------------------------------------------------
// SonarCloud exclusion parsing
// ---------------------------------------------------------------------------

function parseSonarExclusions() {
  try {
    const props = path.join(process.cwd(), 'sonar-project.properties');
    if (!fs.existsSync(props)) return [];
    const content = fs.readFileSync(props, 'utf8');
    for (const line of content.split('\n')) {
      const t = line.trim();
      if (t.startsWith('sonar.exclusions=') || t.startsWith('sonar.coverage.exclusions=')) {
        const val = t.substring(t.indexOf('=') + 1);
        return val.split(',').map(p => p.trim()).filter(Boolean);
      }
    }
    return [];
  } catch {
    return [];
  }
}

function matchesExclusion(filePath, patterns) {
  for (const pattern of patterns) {
    const re = new RegExp(
      '^' + pattern.replace(/\*\*/g, '.*').replace(/\*/g, '[^/]*').replace(/\./g, '\\.') + '$'
    );
    if (re.test(filePath)) return true;
  }
  return false;
}

function shouldAnalyze(file, exclusions) {
  if (file.includes('__mocks__/')) return false;
  if (file.endsWith('.d.ts')) return false;
  if (file.endsWith('.stories.tsx') || file.endsWith('.stories.js')) return false;
  if (file.endsWith('.snap')) return false;
  if (file.endsWith('/types.ts') || file.endsWith('.types.ts')) return false;
  if (file.endsWith('/index.ts') || file.endsWith('/index.js')) return false; // re-export barrels
  if (file.startsWith('development/')) return false; // build tooling
  if (file.startsWith('test/')) return false; // test infrastructure
  if (file.endsWith('.js') && file.includes('scripts/ui')) return false; // entry points
  if (exclusions.length > 0 && matchesExclusion(file, exclusions)) return false;
  return true;
}

// ---------------------------------------------------------------------------
// Jest + LCOV
// ---------------------------------------------------------------------------

function runTestsWithCoverage(testFiles) {
  const args = testFiles.join(' ');
  // Force lcov reporter — extension config only outputs html+json by default
  const cmd = `yarn jest ${args} --coverage --coverageReporters=lcov --passWithNoTests --silent`;

  try {
    console.log(`Running ${testFiles.length} test file(s)...`);
    execSync(cmd, { cwd: process.cwd(), stdio: 'pipe' });
    console.log('All tests passed.\n');
    return { allPassed: true, failedTests: [], passedTests: testFiles };
  } catch (error) {
    const output = (error.stderr || error.stdout || '').toString();
    const passed = [];
    const failed = [];

    for (const line of output.split('\n')) {
      const pm = line.match(/^PASS\s+(.+\.test\.(ts|tsx|js))/);
      const fm = line.match(/^FAIL\s+(.+\.test\.(ts|tsx|js))/);
      if (pm) passed.push(pm[1]);
      if (fm) failed.push({ file: fm[1], error: 'Test failed' });
    }

    console.log(`${passed.length} passed, ${failed.length} failed.`);

    // Re-run passing tests for coverage
    if (passed.length > 0) {
      console.log(`Re-running ${passed.length} passing tests for coverage...\n`);
      try {
        execSync(
          `yarn jest ${passed.join(' ')} --coverage --coverageReporters=lcov --passWithNoTests --silent`,
          { cwd: process.cwd(), stdio: 'pipe' }
        );
      } catch { /* best effort */ }
    }

    return { allPassed: false, failedTests: failed, passedTests: passed };
  }
}

function parseLCOV(sourceFiles) {
  const lcovPath = path.join(COVERAGE_DIR, 'lcov.info');
  if (!fs.existsSync(lcovPath)) {
    // Try root coverage path (sonar config)
    const altPath = path.join(process.cwd(), 'coverage', 'lcov.info');
    if (!fs.existsSync(altPath)) return [];
    // use alt
    return parseLCOVFile(altPath, sourceFiles);
  }
  return parseLCOVFile(lcovPath, sourceFiles);
}

function parseLCOVFile(lcovPath, sourceFiles) {
  const data = fs.readFileSync(lcovPath, 'utf8');
  const results = [];

  for (const record of data.split('end_of_record')) {
    const lines = record.trim().split('\n');
    let filePath = null;
    let totalLines = 0;
    let coveredLines = 0;
    const lineData = {};

    for (const line of lines) {
      if (line.startsWith('SF:')) filePath = line.substring(3);
      else if (line.startsWith('LF:')) totalLines = parseInt(line.substring(3));
      else if (line.startsWith('LH:')) coveredLines = parseInt(line.substring(3));
      else if (line.startsWith('DA:')) {
        const [num, hits] = line.substring(3).split(',').map(Number);
        lineData[num] = hits;
      }
    }

    if (!filePath) continue;
    const rel = path.relative(process.cwd(), filePath);
    if (!sourceFiles.includes(rel)) continue;

    const pct = totalLines > 0 ? Math.round((coveredLines / totalLines) * 100) : 0;
    const changedLines = getChangedLines(rel);

    let newCovered = 0;
    const executableChanged = [];
    for (const ln of changedLines) {
      if (ln in lineData) {
        executableChanged.push(ln);
        if (lineData[ln] > 0) newCovered++;
      }
    }

    const newTotal = executableChanged.length;
    const newPct = newTotal > 0 ? Math.round((newCovered / newTotal) * 100) : 0;

    const uncoveredNewLines = executableChanged.filter(ln => !lineData[ln] || lineData[ln] === 0);

    results.push({
      file: rel,
      coverage: { lines: pct, coveredLines, totalLines, newLines: newPct, newCovered, newTotal, uncoveredNewLines },
    });
  }

  return results;
}

// ---------------------------------------------------------------------------
// Report
// ---------------------------------------------------------------------------

function printReport(coverageResults, noTestFiles, failedTests) {
  const totalCovered = coverageResults.reduce((s, r) => s + r.coverage.coveredLines, 0);
  const totalLines = coverageResults.reduce((s, r) => s + r.coverage.totalLines, 0);
  const overall = totalLines > 0 ? Math.round((totalCovered / totalLines) * 100) : 0;

  const totalNewCovered = coverageResults.reduce((s, r) => s + r.coverage.newCovered, 0);
  const totalNew = coverageResults.reduce((s, r) => s + r.coverage.newTotal, 0);
  const newOverall = totalNew > 0 ? Math.round((totalNewCovered / totalNew) * 100) : 0;

  const belowTarget = coverageResults.filter(r => r.coverage.lines < COVERAGE_THRESHOLD);
  const newBelowTarget = coverageResults.filter(r => r.coverage.newTotal > 0 && r.coverage.newLines < COVERAGE_THRESHOLD);

  console.log('\n--- Coverage Analysis ---');
  console.log(`Overall: ${overall}% (${totalCovered}/${totalLines} lines) | ${coverageResults.length - belowTarget.length}/${coverageResults.length} files >= ${COVERAGE_THRESHOLD}%`);
  console.log(`New code: ${newOverall}% (${totalNewCovered}/${totalNew} changed lines)\n`);

  if (coverageResults.length > 0) {
    console.log('Per-file:');
    for (const r of coverageResults) {
      const icon = r.coverage.lines >= COVERAGE_THRESHOLD ? 'OK' : 'LOW';
      const newInfo = r.coverage.newTotal > 0
        ? `| New: ${r.coverage.newLines}% (${r.coverage.newCovered}/${r.coverage.newTotal})`
        : '| New: no changes';
      const uncoveredInfo = r.coverage.uncoveredNewLines && r.coverage.uncoveredNewLines.length > 0
        ? ` | UNCOVERED LINES: ${r.coverage.uncoveredNewLines.join(',')}`
        : '';
      console.log(`  [${icon}] ${r.file}: ${r.coverage.lines}% (${r.coverage.coveredLines}/${r.coverage.totalLines}) ${newInfo}${uncoveredInfo}`);
    }
  }

  if (noTestFiles.length > 0) {
    console.log(`\nFiles with NO tests (${noTestFiles.length}):`);
    for (const f of noTestFiles.slice(0, 10)) {
      console.log(`  [MISSING] ${f}`);
    }
  }

  if (failedTests.length > 0) {
    console.log(`\nFailed tests (${failedTests.length}):`);
    for (const t of failedTests) {
      console.log(`  [FAIL] ${t.file}`);
    }
  }

  // Summary verdict — only FAIL on actionable blockers (new code uncovered or tests failing)
  const blockers = [];
  const warnings = [];
  if (newBelowTarget.length > 0) blockers.push(`${newBelowTarget.length} file(s) new code below ${COVERAGE_THRESHOLD}%`);
  if (failedTests.length > 0) blockers.push(`${failedTests.length} test(s) failing`);
  if (noTestFiles.length > 0) warnings.push(`${noTestFiles.length} file(s) have no tests`);
  if (belowTarget.length > 0) warnings.push(`${belowTarget.length} file(s) overall below ${COVERAGE_THRESHOLD}% (pre-existing)`);

  if (warnings.length > 0) console.log(`\nWARNINGS: ${warnings.join(', ')}`);

  const pass = blockers.length === 0;
  if (pass) {
    console.log(`VERDICT: PASS — new code meets ${COVERAGE_THRESHOLD}% coverage threshold`);
  } else {
    console.log(`VERDICT: FAIL — ${blockers.join(', ')}`);
  }

  return {
    overall,
    newOverall,
    belowTarget: belowTarget.length,
    newBelowTarget: newBelowTarget.length,
    noTests: noTestFiles.length,
    failedTests: failedTests.length,
    pass,
  };
}

function saveReport(report, branch) {
  // Save reports next to this script (wherever it was injected)
  const reportsDir = path.join(path.dirname(__filename), 'reports');
  if (!fs.existsSync(reportsDir)) fs.mkdirSync(reportsDir, { recursive: true });
  const outPath = path.join(reportsDir, `coverage-report-${branch}.json`);
  fs.writeFileSync(outPath, JSON.stringify(report, null, 2));
  console.log(`\nReport saved: ${outPath}`);
}

// ---------------------------------------------------------------------------
// Main
// ---------------------------------------------------------------------------

function main() {
  const args = process.argv.slice(2);
  let specificFiles = null;

  if (args.includes('--help') || args.includes('-h')) {
    console.log('Usage: node scripts/coverage-analyze.js [--files file1.ts file2.ts]');
    console.log('  No args: analyze all files changed vs main');
    console.log('  --files: analyze specific files');
    process.exit(0);
  }

  const filesIdx = args.indexOf('--files');
  if (filesIdx !== -1) {
    specificFiles = args.slice(filesIdx + 1).filter(a => !a.startsWith('-'));
  } else if (args.length > 0 && !args[0].startsWith('-')) {
    specificFiles = args.filter(a => !a.startsWith('-'));
  }

  console.log('Coverage analysis for changed files...\n');

  const exclusions = parseSonarExclusions();
  if (exclusions.length > 0) console.log(`Loaded ${exclusions.length} SonarCloud exclusion patterns`);

  // Get files
  let files;
  if (specificFiles && specificFiles.length > 0) {
    files = specificFiles.filter(f => fs.existsSync(f) && shouldAnalyze(f, exclusions));
    console.log(`Analyzing ${files.length} specified files`);
  } else {
    const all = getChangedFilesVsMain();
    files = all.filter(f => shouldAnalyze(f, exclusions));
    console.log(`Found ${all.length} changed files, ${files.length} after exclusions`);
  }

  if (files.length === 0) {
    console.log('No files to analyze.');
    process.exit(0);
  }

  // Split by test availability
  const withTests = files.filter(f => hasTestFile(f));
  const noTests = files.filter(f => !hasTestFile(f));
  console.log(`${withTests.length} with tests, ${noTests.length} without tests\n`);

  // Run coverage
  let coverageResults = [];
  let failedTests = [];

  if (withTests.length > 0) {
    const testFiles = withTests.map(f => f.replace(/\.(ts|tsx|js|jsx)$/, '.test.$1')).filter(f => fs.existsSync(f));
    const result = runTestsWithCoverage(testFiles);
    failedTests = result.failedTests;
    coverageResults = parseLCOV(withTests);
  }

  // Report
  const branch = getBranchName();
  const verdict = printReport(coverageResults, noTests, failedTests);

  saveReport({
    branch,
    generatedAt: new Date().toISOString(),
    threshold: COVERAGE_THRESHOLD,
    verdict,
    files: coverageResults,
    noTestFiles: noTests,
    failedTests,
  }, branch);

  process.exit(verdict.pass ? 0 : 1);
}

main();
